@app.route("/debug/")
def debug_ticker(ticker):
    try:
        if not os.path.exists(DATA_PATH):
            return jsonify({
                "ticker": ticker,
                "data_path": DATA_PATH,
                "exists": False,
                "error": "latest.json がありません"
            }), 500

        with open(DATA_PATH, "r", encoding="utf-8") as f:
            data = json.load(f)

        # latest.json は「会社名」がキーなので、tickerから探す
        found = None
        for name, info in data.items():
            if name == "_meta":
                continue
            if isinstance(info, dict) and info.get("ticker") == ticker:
                found = {"name": name, **info}
                break

        return jsonify({
            "ticker": ticker,
            "data_path": DATA_PATH,
            "exists": True,
            "meta": data.get("_meta"),
            "found": found,
            "keys": [k for k in data.keys() if k != "_meta"]
        })
    except Exception as e:
        return jsonify({"ticker": ticker, "error": str(e)}), 500
